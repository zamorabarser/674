\begin{definition}
    The $(k,\ell)$\emph{-tensor bundle} over $M$ is defined as
    \[    \mathcal{T} ^{k}_{\ell} (M) : = \underbrace{ TM \otimes \cdots \otimes TM }_{k  \text{ times}} \otimes  \underbrace{ T^{\ast}M \otimes \cdots \otimes T^{\ast}M}_{\ell \text{ times}} .                     \]
    A section in $\Gamma (\mathcal{T}^k_{\ell})$ is called a \emph{tensor of type} $(k, \ell)$.     
\end{definition}
